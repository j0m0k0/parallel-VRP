cmake_minimum_required(VERSION 3.16)
project(parallel_vrp_ga VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)
find_package(TBB QUIET COMPONENTS tbb)
add_executable(parallel-vrp src/main.cpp)
target_link_libraries(parallel-vrp PRIVATE Threads::Threads)
target_compile_options(parallel-vrp PRIVATE -Wall -Wextra -Wpedantic)
if(TBB_FOUND)
  target_compile_definitions(parallel-vrp PRIVATE PARALLEL_VRP_TBB=1)
  target_link_libraries(parallel-vrp PRIVATE TBB::tbb)
else()
  message(STATUS "oneTBB not found; parallel-vrp uses the same CPU stages with std::thread")
endif()

include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  add_executable(parallel-vrp-cuda src/cuda_main.cu)
  set_target_properties(parallel-vrp-cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
else()
  message(STATUS "CUDA compiler not found; parallel-vrp-cuda will not be built")
endif()
